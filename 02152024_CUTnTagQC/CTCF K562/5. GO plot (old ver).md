```R


df <- df %>%
  mutate(
    neglog10 = -log10(FDR),
    category = factor(category, levels = c("Shared peaks","ChIP confident peaks","CUT&Tag only peaks")),
    term = fct_reorder(term, neglog10, .fun = max)
  )

p <- ggplot(df, aes(x = category, y = term)) +
  geom_point(aes(size = count, color = neglog10), alpha = 0.9) +
  scale_color_viridis_c(name = expression(-log[10]("FDR/q"))) +
  scale_size_continuous(name = "Gene count", range = c(2, 7)) +
  theme_classic(base_size = 11) +
  labs(x = NULL, y = NULL) +
  theme_bw() +
  theme(
    panel.grid.major = element_line(color = "gray70", linetype = "dashed", size = 0.3),
    panel.grid.minor = element_line(color = "gray85", linetype = "dashed", size = 0.2),
    plot.title = element_text(hjust = 0.5, face = "bold", size = 14, color = "black"),
    axis.title = element_text(face = "bold", size = 12, color = "black"),
    axis.text = element_text(size = 10, color = "black"),
    legend.title = element_text(face = "bold", color = "black"),
    legend.text = element_text(color = "black")
  )  +
  theme_classic(base_size = 11) +  
  theme(  
    axis.text.x = element_text(angle = 30, hjust = 1),  
    axis.text.y = element_text(size = 9)  
  )
p
ggsave("GO_dotplot_publication.pdf", p, width = 4.2, height = 3.2, useDingbats = FALSE)
ggsave("GO_dotplot_publication.png", p, width = 6.2, height = 3.2, dpi = 300)
```
