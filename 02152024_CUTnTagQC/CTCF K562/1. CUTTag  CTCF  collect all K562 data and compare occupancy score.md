    mpeaks <- read.table("CTCF_K562_ChIP_all_totalPeak_sampleCoverage.matrix",header = F)
    clean.peaks <- mpeaks[,-c(1:3)]

    output <- cbind(peak, sample.count)
    write.table(output, "CTCF_K562_ChIP_all_totalPeak_with_occupancy_score.bed", row.names = F,col.names = F, sep = "\t", quote=F)
    freq <- table(sample.count)
