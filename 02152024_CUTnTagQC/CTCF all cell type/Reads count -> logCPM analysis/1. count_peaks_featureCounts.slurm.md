#!/bin/bash -l

#SBATCH -J ctcf_fcounts

#SBATCH -p standard

#SBATCH -c 16

#SBATCH --mem=32G

#SBATCH -t 12:00:00

#SBATCH -o /path/to/path/to/external_data/ctcf_fcounts.%j.out

#SBATCH -e /path/to/path/to/external_data/ctcf_fcounts.%j.err

#SBATCH --account=cphg_cz3d

#SBATCH --mail-user=contact@example.com

#SBATCH --mail-type=END,FAIL

  

set -euo pipefail

shopt -s nullglob

  

module load miniforge/24.11.3-py3.12

source "$(conda info --base)/etc/profile.d/conda.sh"

conda activate featurecounts

  

module load gcc

module load samtools

  

BAM_DIR="/path/to/path/to/external_data"

BED="/path/to/path/to/external_data/merged_peaks_300bp.bed"

OUTDIR="/path/to/path/to/external_data"

mkdir -p "${OUTDIR}"

  

SAF="${OUTDIR}/merged_peaks_300bp.saf"

awk 'BEGIN{OFS="\t"}{print "peak_"NR,$1,$2+1,$3,"."}' "${BED}" > "${SAF}"

  

PE_BAMS=()

SE_BAMS=()

for f in "${BAM_DIR}"/*.q30.canonical.pos.bam; do

  if [[ $(samtools view -c -f 1 "$f") -gt 0 ]]; then

    PE_BAMS+=("$f")

  else

    SE_BAMS+=("$f")

  fi

done

  

(( ${#PE_BAMS[@]} + ${#SE_BAMS[@]} > 0 )) || { echo "No BAMs found"; exit 1; }

  

# Paired-end counts (fragments)

if (( ${#PE_BAMS[@]} )); then

  featureCounts -T 16 -F SAF -p \

    -a "${SAF}" \

    -o "${OUTDIR}/mergedPeaks300bp.featureCounts.pe.txt" \

    "${PE_BAMS[@]}"

fi

  

# Single-end counts (reads)

if (( ${#SE_BAMS[@]} )); then

  featureCounts -T 16 -F SAF \

    -a "${SAF}" \

    -o "${OUTDIR}/mergedPeaks300bp.featureCounts.se.txt" \

    "${SE_BAMS[@]}"

fi
