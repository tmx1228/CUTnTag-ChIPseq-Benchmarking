### 0. CTCF motif
ctcf_motif=/path/to/lab_data/reference/ctcf/CTCF_map.bed
module load gcc

### 1. shared confident peaks (ChIP score >22 && CUT&Tag score > 2)

sort cuttag.zero.tmp.bed > cuttag.zero.sorted.bed

bedtools intersect -a ChIP_bias.bed -b $ctcf_motif -u > ChIP_bias_with_motif.bed
bedtools intersect -a ChIP_bias.bed -b $ctcf_motif -v > ChIP_bias_no_motif.bed

### 5. CUTTag bias (ChIP score == 0 && CUT&Tag score == 1)
awk 'BEGIN{OFS="\t"} $4 == 0{print $1, $2, $3}' CTCF_K562_ChIP_on_chipCuttagUnionPeak_with_occupancyScore.bed > chip.zero.tmp.bed
awk 'BEGIN{OFS="\t"} $4 == 1{print $1, $2, $3}'  CTCF_K562_cuttag_on_chipCuttagUnionPeak_with_occupancyScore.bed > cuttag.one.tmp.bed
