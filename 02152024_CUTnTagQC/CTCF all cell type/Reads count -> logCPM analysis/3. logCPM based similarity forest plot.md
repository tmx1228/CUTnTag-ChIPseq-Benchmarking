## 2) Sample-to-sample correlation (Pearson)
```R

logCPM_sub <- readRDS("/path/to/path/to/external_data/logCPM_sub_50k.rds")

sim <- cor(logCPM_sub, method = "pearson", use = "pairwise.complete.obs")

## 3) Use upper triangle only (each pair counted once)
ut <- which(upper.tri(sim), arr.ind = TRUE)
idx <- ut                       # Preserve the original idx structure
sim_vec <- sim[ut]

## 4) Fisher z transform（replaces logit_sim）
eps <- 1e-6
sim_vec <- pmin(pmax(sim_vec, -1 + eps), 1 - eps)
z <- atanh(sim_vec)

## 5) df_lm：pairwise table
df_lm <- tibble(
  i = idx[,1],
  j = idx[,2],
  sample1 = samples[i],
  sample2 = samples[j],
  z = z
)

## 6) Merge metadata onto both sides
m1 <- meta2; colnames(m1) <- paste0(colnames(m1), "_1")
m2 <- meta2; colnames(m2) <- paste0(colnames(m2), "_2")

df_lm <- df_lm %>%
  left_join(m1, by = c("sample1" = "GSM_1")) %>%
  left_join(m2, by = c("sample2" = "GSM_2")) %>%
  mutate(
    same_cell      = (cell_1 == cell_2),
    same_tech      = (technique_1 == technique_2),
    same_treatment = (treatment_1 == treatment_2)
  )
```

### jackknife framework for LM (signal version)
```R
library(forcats)
library(ggplot2)

## Original model
fit0 <- lm(z ~ same_cell + same_tech + same_treatment, data = df_lm)
beta0 <- coef(fit0)

## jackknife：Remove all pairs involving one sample per iteration
u <- unique(c(df_lm$sample1, df_lm$sample2))
n <- length(u)

B <- matrix(NA_real_, nrow = n, ncol = length(beta0))
colnames(B) <- names(beta0)

for (k in seq_len(n)) {
  s <- u[k]
  keep_k <- (df_lm$sample1 != s) & (df_lm$sample2 != s)
  fit_k <- lm(z ~ same_cell + same_tech + same_treatment, data = df_lm[keep_k, ])
  B[k, ] <- coef(fit_k)
}

## jackknife SE
beta_bar <- colMeans(B, na.rm = TRUE)
se_jack <- sqrt((n - 1) / n *
                  colSums((B - matrix(beta_bar, n, length(beta0), byrow = TRUE))^2, na.rm = TRUE))

## df = n-1 conservative t-test
crit <- qt(0.975, df = n - 1)

model_coef_jack <- tibble(
  term = names(beta0),
  estimate = as.numeric(beta0),
  std.error = as.numeric(se_jack),
  conf.low = estimate - crit * std.error,
  conf.high = estimate + crit * std.error,
  statistic = estimate / std.error,
  p.value = 2 * pt(abs(statistic), df = n - 1, lower.tail = FALSE)
)

model_coef_jack



write.csv(model_coef_jack, "coef_jack_stats.csv", row.names = FALSE)
```
