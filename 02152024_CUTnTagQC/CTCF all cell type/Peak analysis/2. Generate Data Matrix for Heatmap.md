#### generate top 50k anchors
```R
setwd("/sfs/weka/path/to/path/to/external_data")


library(readr)
library(dplyr)
library(pheatmap)

# Read binary matrix
bin_df <- read_csv(
  "/path/to/path/to/external_data/binary_peak_matrix.csv",
  show_col_types = FALSE
)

bin_mat <- as.data.frame(bin_df)
rownames(bin_mat) <- bin_mat$anchor_id
bin_mat$anchor_id <- NULL
bin_mat <- as.matrix(bin_mat)

# Select the 50,000 most informative rows
row_sum <- rowSums(bin_mat)
keep <- row_sum >= 2 & row_sum <= (ncol(bin_mat) - 2)
mat_f <- bin_mat[keep, , drop = FALSE]

p <- rowSums(mat_f) / ncol(mat_f)
row_var <- p * (1 - p)

top_n <- min(50000, nrow(mat_f))
top_idx <- order(row_var, decreasing = TRUE)[1:top_n]
mat_top50k <- mat_f[top_idx, , drop = FALSE]

saveRDS(
  mat_top50k,
  file = "/path/to/path/to/external_data/binary_peak_matrix_top50000.rds"
)

```
