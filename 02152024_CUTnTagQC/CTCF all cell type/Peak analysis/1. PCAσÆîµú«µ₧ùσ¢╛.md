```R
### Calculate Jaccard

# bin_mat: rows = peaks, columns = samples，0/1
X <- Matrix(as.matrix(bin_mat), sparse = TRUE)

# 1) Pairwise intersection counts
inter <- as.matrix(crossprod(X))   # sample x sample

# 2) Union counts
cs <- Matrix::colSums(X)
uni <- outer(cs, cs, "+") - inter

# 3) Jaccard similarity
J <- inter / pmax(uni, 1)          # Avoid division by zero
diag(J) <- 1

# 4) Keep only upper-triangle sample pairs
idx <- which(upper.tri(J), arr.ind = TRUE)
samples <- colnames(bin_mat)

jaccard_pairs <- tibble(
  sample1 = samples[idx[, 1]],
  sample2 = samples[idx[, 2]],
  similarity = J[idx]
)

# 5) Add metadata (including paired and treatment)
# First clean into a base table to avoid duplicate columns
jaccard_pairs <- jaccard_pairs %>%
  select(sample1, sample2, similarity) %>%
  left_join(
    meta %>% select(id, caller, cell, technique, paired, treatment),
    by = c("sample1" = "id")
  ) %>%
  rename(
    caller1 = caller, cell1 = cell, tech1 = technique,
    paired1 = paired, treatment1 = treatment
  ) %>%
  left_join(
    meta %>% select(id, caller, cell, technique, paired, treatment),
    by = c("sample2" = "id")
  ) %>%
  rename(
    caller2 = caller, cell2 = cell, tech2 = technique,
    paired2 = paired, treatment2 = treatment
  ) %>%
  mutate(
    same_caller = caller1 == caller2,
    same_cell = cell1 == cell2,
    same_tech = tech1 == tech2,
    same_paired = paired1 == paired2,
    same_treatment = treatment1 == treatment2,
    treatment_group1 = ifelse(toupper(trimws(treatment1)) == "WT", "WT", "Other"),
    treatment_group2 = ifelse(toupper(trimws(treatment2)) == "WT", "WT", "Other"),

  )
```

### Generate a new coefficient table with jackknife
```R
library(dplyr)
library(ggplot2)
library(forcats)

# 1) Add sample IDs for both ends of each pair to df_lm
df_lm$sample1 <- samples[idx[, 1]]
df_lm$sample2 <- samples[idx[, 2]]

# 2) Original model（consistent with the fitted model）
fit0 <- lm(
  logit_sim ~ same_cell + same_tech + same_caller + same_treatment_group,
  data = df_lm
)
beta0 <- coef(fit0)

# 3) Leave-one-sample-out jackknife
u <- unique(c(df_lm$sample1, df_lm$sample2))
n <- length(u)

B <- matrix(NA_real_, nrow = n, ncol = length(beta0))
colnames(B) <- names(beta0)

for (k in seq_len(n)) {
  s <- u[k]
  keep <- (df_lm$sample1 != s) & (df_lm$sample2 != s)

  fit_k <- lm(
    logit_sim ~ same_cell + same_tech + same_caller + same_treatment_group,
    data = df_lm[keep, ]
  )
  B[k, ] <- coef(fit_k)
}

beta_bar <- colMeans(B, na.rm = TRUE)
se_jack <- sqrt((n - 1) / n * colSums((B - matrix(beta_bar, n, length(beta0), byrow = TRUE))^2, na.rm = TRUE))

# More conservative: use df = n-1 (reasonable when n=366)
crit <- qt(0.975, df = n - 1)

model_coef_jack <- tibble(
  term = names(beta0),
  estimate = as.numeric(beta0),
  std.error = as.numeric(se_jack),
  conf.low = estimate - crit * std.error,
  conf.high = estimate + crit * std.error,
  statistic = estimate / std.error,
  p.value = 2 * pt(abs(statistic), df = n - 1, lower.tail = FALSE)
)
```
