```shell
# Generate length table (without overlap)
: > peak_lengths.tsv
for f in *.narrowPeak *.scoreisland *.stringent.bed; do
  [ -e "$f" ] || continue

  case "$f" in
    *_peaks.narrowPeak) caller="macs2" ;;
    *.scoreisland)      caller="sicer" ;;
    *.stringent.bed)    caller="seacr" ;;
    *)                  caller="unknown" ;;
  esac

  # Extract GSM IDs from filenames (robust matching: GSM followed by digits)
  gsm=$(echo "$f" | grep -oE 'GSM[0-9]+' | head -n 1)

  awk -v OFS='\t' -v file="$f" -v caller="$caller" -v gsm="$gsm" '
    $2 < $3 {
      id = gsm "_" caller
      print id, $1, $2, $3, $3-$2
    }
  ' "$f" >> peak_lengths.tsv
done
```
