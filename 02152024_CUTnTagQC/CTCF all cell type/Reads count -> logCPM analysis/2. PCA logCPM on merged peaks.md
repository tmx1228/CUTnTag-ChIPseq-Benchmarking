### Merge PE and SE read signal matrices on union peaks
```R
library(data.table)

read_fc <- function(file){

  dt <- fread(file, skip = 1)

  counts <- dt[, -(1:6)]

  cn <- basename(colnames(counts))

  cn <- sub("_.*", "", cn)

  colnames(counts) <- cn

  as.matrix(counts)
}

count_pe <- read_fc("mergedPeaks300bp.featureCounts.pe.txt")

count_se <- read_fc("mergedPeaks300bp.featureCounts.se.txt")

count_all <- cbind(count_pe, count_se)

# Check for duplicate GSM column names (important!)  
anyDuplicated(colnames(count_all))
```

### Filter data and calculate logCPM
```R
# Filter: counts >=5 in at least two samples
keep_peaks <- rowSums(count_all >= 5) >= 2
count_f <- count_all[keep_peaks, , drop = FALSE]

libsize <- colSums(count_f)
cpm <- t(t(count_f) / libsize * 1e6)
logCPM <- log2(cpm + 1)

dim(logCPM)

saveRDS(  
logCPM,  
file = "/path/to/path/to/external_data/logCPM_merged_peaks_122samples.rds"  
)
```

### PCA by cell type and technique
```R
library(ggplot2)

# PCA（logCPM has already been calculated）
pca <- prcomp(t(logCPM), scale. = TRUE)

pca_df <- data.frame(
  GSM = rownames(pca$x),
  PC1 = pca$x[,1],
  PC2 = pca$x[,2]
)

head(pca_df)
pca_df <- merge(pca_df, as.data.frame(meta2), by = "GSM", all.x = TRUE)


# Ensure stable factor ordering (optional)
pca_df$paired <- factor(pca_df$paired, levels = c("PE","SE"))
pca_df$technique <- factor(pca_df$technique)
pca_df$cell <- factor(pca_df$cell)
saveRDS(pca_df, "pca_df.rds")
```
